cmake_minimum_required(VERSION 3.16)
project(so101_mtc_description NONE)

find_package(ament_cmake REQUIRED)
find_package(xacro REQUIRED)

# Third-party models (ADR 0002): tools/fetch_models.py fetches them, unmodified, into the workspace's src/, next to
# this repository. They are installed under share/${PROJECT_NAME}/upstream/, so that the URDF, the launch files and
# MuJoCo find them by package name; with colcon's --symlink-install the installed files are links to the fetched ones.
get_filename_component(MODELS "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SO101 "${MODELS}/SO-ARM100/Simulation/SO101")
set(MENAGERIE_SO101 "${MODELS}/mujoco_menagerie/robotstudio_so101")
foreach(required "${SO101}/so101_new_calib.urdf" "${MENAGERIE_SO101}/so101.xml")
  if(NOT EXISTS "${required}")
    message(FATAL_ERROR "${required} is missing. Fetch the pinned models first, from the workspace: "
      "python3 src/so101_mtc/tools/fetch_models.py")
  endif()
endforeach()

# The official URDF points to its meshes relative to itself (filename="assets/..."), which RViz and MoveIt cannot
# resolve. The installed copy points to them by package (package://...); the fetched file stays unmodified.
set(MESHES "package://${PROJECT_NAME}/upstream/SO101/assets/")
file(READ "${SO101}/so101_new_calib.urdf" urdf)
string(REPLACE "filename=\"assets/" "filename=\"${MESHES}" urdf "${urdf}")
string(FIND "${urdf}" "filename=\"${MESHES}" rewritten)
string(FIND "${urdf}" "filename=\"assets/" left)
if(rewritten EQUAL -1 OR NOT left EQUAL -1)
  message(FATAL_ERROR "The mesh paths of ${SO101}/so101_new_calib.urdf changed: update the rewrite above")
endif()
string(REPLACE "<robot " "<!-- Generated by so101_mtc_description/CMakeLists.txt from SO-ARM100 \
Simulation/SO101/so101_new_calib.urdf: the same file with its mesh paths as package:// URIs. Do not edit. -->\n<robot "
  urdf "${urdf}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/so101_new_calib.urdf" "${urdf}")
# a new pin of the model changes the fetched file: configure again
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SO101}/so101_new_calib.urdf")

# MJCF has no variables: the scene is a xacro template, expanded here with the block of config/task.yaml and
# installed as mujoco/scene.xml. xacro tracks task.yaml as a dependency, so editing it rebuilds the scene.
xacro_add_files(mujoco/scene.xml.xacro REMAP "task:=${CMAKE_CURRENT_SOURCE_DIR}/config/task.yaml"
  TARGET scene INSTALL DESTINATION mujoco)

install(DIRECTORY config launch rviz urdf DESTINATION share/${PROJECT_NAME})
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/so101_new_calib.urdf" "${MODELS}/SO-ARM100/LICENSE"
  DESTINATION share/${PROJECT_NAME}/upstream/SO101)
install(DIRECTORY "${SO101}/assets" DESTINATION share/${PROJECT_NAME}/upstream/SO101)
install(FILES "${MENAGERIE_SO101}/so101.xml" "${MENAGERIE_SO101}/LICENSE"
  DESTINATION share/${PROJECT_NAME}/upstream/robotstudio_so101)
install(DIRECTORY "${MENAGERIE_SO101}/assets" DESTINATION share/${PROJECT_NAME}/upstream/robotstudio_so101)

if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  # the installed package: xacro → URDF, the MuJoCo scene, and their agreement
  ament_add_pytest_test(test_description test/test_description.py TIMEOUT 120)
endif()

ament_package()
